# meds-api — aucune indexation, aucun entraînement de modèle.
#
# Cette API n'expose librement que /docs et /openapi.json ; tout le reste exige
# une clé. Ces deux routes n'ont pas vocation à être indexées, ni à servir de
# corpus d'entraînement ou de source de récupération à la volée.
#
# Les données servies proviennent de la BDPM, publiée en open data par l'ANSM.
# Il faut les récupérer à la source plutôt que moissonner cette API :
#   https://base-donnees-publique.medicaments.gouv.fr/
#
# Voir aussi /ai.txt, /llms.txt et l'en-tête de réponse X-Robots-Tag.
# Les robots d'entraînement notoires reçoivent en outre un 403 au frontal —
# mais jamais sur ce fichier : la RFC 9309 §2.3.1.4 interprète un robots.txt
# en 4xx comme « aucune restriction », ce qui produirait l'effet inverse.

# ─── Règle générale (RFC 9309) ───────────────────────────────────────────────
# Elle suffit à tout robot conforme. Les sections nommées plus bas ne sont là
# que pour ceux qui n'appliquent que la leur.

User-agent: *
Disallow: /

# Content Signals Policy : refus explicite des trois usages en aval.
Content-Signal: search=no, ai-input=no, ai-train=no

# d'Applebot, qui alimente Siri et Spotlight.
